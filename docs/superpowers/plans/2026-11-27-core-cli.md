# Core CLI Implementation Plan (plan 1 of 4)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** An `adf` command that builds a warm Linux desktop template on smolvm, branches cheap desktops from it, drives them (screenshot, click, type, key, scroll, drag) and removes them, on macOS first and Linux KVM second.

**Architecture:** One Go module. `internal/smolvm` wraps the `smolvm` CLI (JSON output, JSON errors). `internal/fleet` owns the template and desktops and depends only on a small `Runtime` interface that `*smolvm.Client` satisfies. `internal/computer` turns computer-use actions into `xdotool` and `scrot` commands executed inside a desktop. `cmd/adf` is a thin cobra CLI over these packages. Screenshots and input run inside the guest because smolvm 1.37.7 branches get no host display frames (smol-machines/smolvm#2475).

**Tech Stack:** Go 1.25, github.com/spf13/cobra, smolvm 1.37.7 (libkrun), Debian trixie guest with Xvfb, openbox, tint2, xdotool, scrot, Firefox ESR.

**Later plans (not in scope here):** 2. MCP server (stdio and streamable HTTP) with lifecycle and computer tools. 3. Daemon, REST API and web console with a live thumbnail grid and takeover. 4. Benchmark harness, Linux packaging and notarized releases.

**Facts this plan relies on (verified 2026-11-27 on smolvm 1.37.7, macOS 27.2.1 M5):**
- `smolvm machine ls --json` prints an array with `name`, `state`, `pid` (null when stopped), `labels`, `branchable`, `parent_machine`, `image`, `cpus`, `memory_mib`, `created_at`.
- `--error-format json` makes smolvm print `{"error":{"code":"NOT_FOUND","message":"..."}}` on stderr for its own failures.
- `machine exec` passes the guest command's stdout (binary safe), stderr and exit code straight through. `-i` forwards stdin.
- `machine branch --from SRC --name CHILD --label K=V` works on a source started with `machine start --branchable`. Children inherit the source's labels; `--label` overrides a key.
- `machine delete --force` skips the prompt; a branch source with children needs `--cascade`.
- Inside the guest: `xdotool key -- ctrl+l`, `xdotool type --delay 12 -- "-n hello"`, `click --repeat 2 --delay 80 1`, `mousedown 1 ... mouseup 1`, `getmouselocation --shell` and `scrot -o -z FILE` (scrot 1.12.1) all work. Do not use `mousemove --sync`; it can hang when the pointer is already at the target.

// Package runner runs host commands. It exists so tests can replace real processes.
package runner

import (
	"bytes"
	"context"
	"errors"
	"fmt"
	"io"
	"os/exec"
	"strings"
)

// Runner runs a program with optional stdin and returns its stdout.
type Runner interface {
	Run(ctx context.Context, stdin io.Reader, name string, args ...string) ([]byte, error)
}

// ExitError is returned when the program ran but exited non-zero.
type ExitError struct {
	Name   string
	Args   []string
	Code   int
	Stderr []byte
}

func (e *ExitError) Error() string {
	msg := strings.TrimSpace(string(e.Stderr))
	if msg == "" {
		msg = "no output on stderr"
	}
	return fmt.Sprintf("%s exited with code %d: %s", e.Name, e.Code, msg)
}

// Exec runs real processes.
type Exec struct{}

// Run implements Runner.
func (Exec) Run(ctx context.Context, stdin io.Reader, name string, args ...string) ([]byte, error) {
	cmd := exec.CommandContext(ctx, name, args...)
	cmd.Stdin = stdin
	var stdout, stderr bytes.Buffer
	cmd.Stdout = &stdout
	cmd.Stderr = &stderr
	err := cmd.Run()
	if err == nil {
		return stdout.Bytes(), nil
	}
	var exitErr *exec.ExitError
	if errors.As(err, &exitErr) {
		return stdout.Bytes(), &ExitError{Name: name, Args: args, Code: exitErr.ExitCode(), Stderr: stderr.Bytes()}
	}
	return nil, fmt.Errorf("run %s: %w", name, err)
}
```

import (

// Call is one recorded Fake.Run invocation.
type Call struct {
	Name  string
	Args  []string
	Stdin []byte
}

// Result is one scripted Fake.Run outcome.
type Result struct {
	Stdout []byte
	Err    error
}

// Fake records calls and replays Results in order. Once Results is empty it returns (nil, nil).
type Fake struct {
	Calls   []Call
	Results []Result
}

// Run implements Runner.
func (f *Fake) Run(_ context.Context, stdin io.Reader, name string, args ...string) ([]byte, error) {
	c := Call{Name: name, Args: append([]string(nil), args...)}
	if stdin != nil {
		b, err := io.ReadAll(stdin)
		if err != nil {
			return nil, err
		}
		c.Stdin = b
	}
	f.Calls = append(f.Calls, c)
	if len(f.Results) == 0 {
		return nil, nil
	}
	r := f.Results[0]
	f.Results = f.Results[1:]
	return r.Stdout, r.Err
}

package smolvm

import (
	"encoding/json"
	"errors"
	"strings"

	"github.com/CodeGeek04/agent-desktop-fleet/internal/runner"
)

// Error is a failure reported by smolvm itself (not by a command running inside a machine).
type Error struct {
	Code    string
	Message string
}

func (e *Error) Error() string { return "smolvm: " + e.Code + ": " + e.Message }

// IsNotFound reports whether err is smolvm's NOT_FOUND error.
func IsNotFound(err error) bool {
	var e *Error
	return errors.As(err, &e) && e.Code == "NOT_FOUND"
}

// parseError turns smolvm's JSON error line into *Error. Anything else is returned unchanged,
// which keeps a guest command's own exit code and stderr visible to callers.
func parseError(err error) error {
	var ee *runner.ExitError
	if !errors.As(err, &ee) {
		return err
	}
	lines := strings.Split(strings.TrimSpace(string(ee.Stderr)), "\n")
	last := strings.TrimSpace(lines[len(lines)-1])
	var payload struct {
		Error *struct {
			Code    string `json:"code"`
			Message string `json:"message"`
		} `json:"error"`
	}
	if json.Unmarshal([]byte(last), &payload) == nil && payload.Error != nil && payload.Error.Code != "" {
		return &Error{Code: payload.Error.Code, Message: payload.Error.Message}
	}
	return err
}

// Package smolvm drives the smolvm CLI.
package smolvm

import (
	"context"
	"encoding/json"
	"fmt"
	"io"

	"github.com/CodeGeek04/agent-desktop-fleet/internal/runner"
)

// Machine is one entry of `smolvm machine ls --json`.
type Machine struct {
	Name          string            `json:"name"`
	State         string            `json:"state"`
	PID           int               `json:"pid"`
	Image         string            `json:"image"`
	CPUs          int               `json:"cpus"`
	MemoryMiB     int               `json:"memory_mib"`
	Labels        map[string]string `json:"labels"`
	Branchable    bool              `json:"branchable"`
	ParentMachine string            `json:"parent_machine"`
	CreatedAt     int64             `json:"created_at"`
}

// Client runs smolvm commands.
type Client struct {
	Bin    string
	Runner runner.Runner
}

// New returns a client for the smolvm binary at bin.
func New(bin string, r runner.Runner) *Client { return &Client{Bin: bin, Runner: r} }

// machine runs `smolvm machine SUB --error-format json ARGS...`.
func (c *Client) machine(ctx context.Context, stdin io.Reader, sub string, args ...string) ([]byte, error) {
	full := append([]string{"machine", sub, "--error-format", "json"}, args...)
	out, err := c.Runner.Run(ctx, stdin, c.Bin, full...)
	if err != nil {
		return out, parseError(err)
	}
	return out, nil
}

// List returns all machines.
func (c *Client) List(ctx context.Context) ([]Machine, error) {
	out, err := c.machine(ctx, nil, "ls", "--json")
	if err != nil {
		return nil, err
	}
	var ms []Machine
	if err := json.Unmarshal(out, &ms); err != nil {
		return nil, fmt.Errorf("parse smolvm machine ls: %w", err)
	}
	return ms, nil
}

// CreateOpts configures a new machine.
type CreateOpts struct {
	Name       string
	Image      string
	CPUs       int
	MemoryMiB  int
	OverlayGiB int
	Net        bool
	Labels     map[string]string
}

// BranchOpts configures a branch of a running branchable machine.
type BranchOpts struct {
	From   string
	Name   string
	Labels map[string]string
}

// labelArgs returns --label flags in a stable order.
func labelArgs(labels map[string]string) []string {
	keys := make([]string, 0, len(labels))
	for k := range labels {
		keys = append(keys, k)
	}
	sort.Strings(keys)
	var args []string
	for _, k := range keys {
		args = append(args, "--label", k+"="+labels[k])
	}
	return args
}

// Create defines a machine without starting it.
func (c *Client) Create(ctx context.Context, o CreateOpts) error {
	args := []string{"--name", o.Name, "--image", o.Image,
		"--cpus", strconv.Itoa(o.CPUs), "--mem", strconv.Itoa(o.MemoryMiB), "--overlay", strconv.Itoa(o.OverlayGiB)}
	if o.Net {
		args = append(args, "--net")
	}
	args = append(args, labelArgs(o.Labels)...)
	_, err := c.machine(ctx, nil, "create", args...)
	return err
}

// Start boots a machine. A branchable machine can be branched while it runs.
func (c *Client) Start(ctx context.Context, name string, branchable bool) error {
	args := []string{"--name", name}
	if branchable {
		args = append(args, "--branchable")
	}
	_, err := c.machine(ctx, nil, "start", args...)
	return err
}

// Branch creates a running copy-on-write child of a running branchable machine.
func (c *Client) Branch(ctx context.Context, o BranchOpts) error {
	args := append([]string{"--from", o.From, "--name", o.Name}, labelArgs(o.Labels)...)
	_, err := c.machine(ctx, nil, "branch", args...)
	return err
}

// Stop stops a machine.
func (c *Client) Stop(ctx context.Context, name string) error {
	_, err := c.machine(ctx, nil, "stop", "--name", name)
	return err
}

// Delete removes a machine. cascade also removes children branched from it.
func (c *Client) Delete(ctx context.Context, name string, cascade bool) error {
	args := []string{"--name", name, "--force"}
	if cascade {
		args = append(args, "--cascade")
	}
	_, err := c.machine(ctx, nil, "delete", args...)
	return err
}
```

// Package computer turns computer-use actions into commands that run inside a desktop.
package computer

import (
	"context"
	"errors"
	"fmt"
	"io"
	"regexp"
	"strconv"
	"strings"
)

// Execer runs argv inside one desktop and returns its stdout.
type Execer func(ctx context.Context, stdin io.Reader, argv ...string) ([]byte, error)

// Screen describes the desktop's X display.
type Screen struct {
	Display string
	Width   int
	Height  int
}

// Button is a mouse button.
type Button string

// Mouse buttons. The empty Button means Left.
const (
	Left   Button = "left"
	Middle Button = "middle"
	Right  Button = "right"
)

// Direction is a scroll direction.
type Direction string

// Scroll directions.
const (
	ScrollUp    Direction = "up"
	ScrollDown  Direction = "down"
	ScrollLeft  Direction = "left"
	ScrollRight Direction = "right"
)

// MaxScroll is the largest number of wheel clicks in one Scroll call.
const MaxScroll = 50

// Computer performs actions on one desktop.
type Computer struct {
	exec   Execer
	screen Screen
}

// New returns a Computer that runs commands through exec.
func New(exec Execer, screen Screen) *Computer { return &Computer{exec: exec, screen: screen} }

func (c *Computer) xdotool(ctx context.Context, args ...string) ([]byte, error) {
	argv := append([]string{"env", "DISPLAY=" + c.screen.Display, "xdotool"}, args...)
	return c.exec(ctx, nil, argv...)
}

func (c *Computer) checkPoint(x, y int) error {
	if x < 0 || y < 0 || x >= c.screen.Width || y >= c.screen.Height {
		return fmt.Errorf("point (%d, %d) is outside the %dx%d screen", x, y, c.screen.Width, c.screen.Height)
	}
	return nil
}

func buttonNumber(b Button) (string, error) {
	switch b {
	case Left, "":
		return "1", nil
	case Middle:
		return "2", nil
	case Right:
		return "3", nil
	}
	return "", fmt.Errorf("unknown mouse button %q (want left, middle or right)", b)
}

func itoa(i int) string { return strconv.Itoa(i) }

// Move moves the pointer.
func (c *Computer) Move(ctx context.Context, x, y int) error {
	if err := c.checkPoint(x, y); err != nil {
		return err
	}
	_, err := c.xdotool(ctx, "mousemove", itoa(x), itoa(y))
	return err
}

// Click moves the pointer and clicks once.
func (c *Computer) Click(ctx context.Context, x, y int, b Button) error {
	if err := c.checkPoint(x, y); err != nil {
		return err
	}
	n, err := buttonNumber(b)
	if err != nil {
		return err
	}
	_, err = c.xdotool(ctx, "mousemove", itoa(x), itoa(y), "click", n)
	return err
}

// DoubleClick moves the pointer and clicks twice.
func (c *Computer) DoubleClick(ctx context.Context, x, y int, b Button) error {
	if err := c.checkPoint(x, y); err != nil {
		return err
	}
	n, err := buttonNumber(b)
	if err != nil {
		return err
	}
	_, err = c.xdotool(ctx, "mousemove", itoa(x), itoa(y), "click", "--repeat", "2", "--delay", "80", n)
	return err
}

// Drag presses the left button at (x1, y1), moves to (x2, y2) and releases.
func (c *Computer) Drag(ctx context.Context, x1, y1, x2, y2 int) error {
	if err := c.checkPoint(x1, y1); err != nil {
		return err
	}
	if err := c.checkPoint(x2, y2); err != nil {
		return err
	}
	_, err := c.xdotool(ctx, "mousemove", itoa(x1), itoa(y1), "mousedown", "1",
		"mousemove", itoa(x2), itoa(y2), "mouseup", "1")
	return err
}

// Scroll moves the pointer and turns the wheel amount clicks in dir.
func (c *Computer) Scroll(ctx context.Context, x, y int, dir Direction, amount int) error {
	if err := c.checkPoint(x, y); err != nil {
		return err
	}
	if amount < 1 || amount > MaxScroll {
		return fmt.Errorf("scroll amount %d out of range 1 to %d", amount, MaxScroll)
	}
	buttons := map[Direction]string{ScrollUp: "4", ScrollDown: "5", ScrollLeft: "6", ScrollRight: "7"}
	b, ok := buttons[dir]
	if !ok {
		return fmt.Errorf("unknown scroll direction %q (want up, down, left or right)", dir)
	}
	_, err := c.xdotool(ctx, "mousemove", itoa(x), itoa(y), "click", "--repeat", itoa(amount), b)
	return err
}

// Type types text as keyboard input.
func (c *Computer) Type(ctx context.Context, text string) error {
	if text == "" {
		return nil
	}
	_, err := c.xdotool(ctx, "type", "--delay", "12", "--", text)
	return err
}

// keyRE allows xdotool key names and chords such as Return, ctrl+l, ctrl+shift+Tab, F5.
var keyRE = regexp.MustCompile(`^[A-Za-z0-9_]+(\+[A-Za-z0-9_]+)*$`)

// Key presses a key or chord such as "Return" or "ctrl+l". Several keys separated by
// spaces are pressed in order.
func (c *Computer) Key(ctx context.Context, combo string) error {
	keys := strings.Fields(combo)
	if len(keys) == 0 {
		return errors.New("no key given")
	}
	for _, k := range keys {
		if !keyRE.MatchString(k) {
			return fmt.Errorf("invalid key %q", k)
		}
	}
	_, err := c.xdotool(ctx, append([]string{"key", "--"}, keys...)...)
	return err
}

var nameRE = regexp.MustCompile(`^[a-z0-9]([a-z0-9-]{0,38}[a-z0-9])?$`)

// ValidateName checks a user-facing desktop name.
func ValidateName(name string) error {
	if !nameRE.MatchString(name) {
		return fmt.Errorf("invalid desktop name %q: use 1 to 40 lowercase letters, digits or dashes, not starting or ending with a dash", name)
	}
	if name == "template" {
		return errors.New(`"template" is a reserved name`)
	}
	return nil
}

func (m *Manager) logf(format string, args ...any) {
	if m.Log != nil {
		fmt.Fprintf(m.Log, format+"\n", args...)
	}
}

func (m *Manager) find(ctx context.Context, machine string) (*smolvm.Machine, error) {
	ms, err := m.RT.List(ctx)
	if err != nil {
		return nil, err
	}
	for i := range ms {
		if ms[i].Name == machine {
			return &ms[i], nil
		}
	}
	return nil, nil
}

// ensureBranchSource leaves the template running with --branchable.
func (m *Manager) ensureBranchSource(ctx context.Context, t *smolvm.Machine) error {
	if t.State == "running" && t.Branchable {
		return nil
	}
	if t.State == "running" {
		if err := m.RT.Stop(ctx, TemplateName); err != nil {
			return fmt.Errorf("stop template: %w", err)
		}
	}
	m.logf("starting template (the first start downloads the base image)")
	if err := m.RT.Start(ctx, TemplateName, true); err != nil {
		return fmt.Errorf("start template: %w", err)
	}
	return nil
}

// EnsureTemplate creates and provisions the template if needed and leaves it running as a branch source.
func (m *Manager) EnsureTemplate(ctx context.Context) error {
	t, err := m.find(ctx, TemplateName)
	if err != nil {
		return err
	}
	if t == nil {
		m.logf("creating template %s from %s", TemplateName, m.Image)
		err := m.RT.Create(ctx, smolvm.CreateOpts{
			Name: TemplateName, Image: m.Image, CPUs: 2, MemoryMiB: 2048, OverlayGiB: 8, Net: true,
			Labels: map[string]string{LabelRole: RoleTemplate},
		})
		if err != nil {
			return fmt.Errorf("create template: %w", err)
		}
		t = &smolvm.Machine{Name: TemplateName, State: "stopped"}
	}
	if err := m.ensureBranchSource(ctx, t); err != nil {
		return err
	}
	out, err := m.RT.Exec(ctx, TemplateName, nil, "cat", "/etc/adf-image-version")
	if err == nil && strings.TrimSpace(string(out)) == image.Version {
		return nil
	}
	m.logf("installing desktop packages (one time, about 1 to 2 minutes)")
	steps := []struct {
		stdin string
		argv  []string
	}{
		{image.SetupScript, []string{"sh", "-c", "cat > /tmp/adf-setup.sh"}},
		{"", []string{"sh", "/tmp/adf-setup.sh"}},
		{image.StartScript, []string{"sh", "-c", "cat > /usr/local/bin/desktop-start && chmod 755 /usr/local/bin/desktop-start"}},
		{"", []string{"sh", "-c", "echo " + image.Version + " > /etc/adf-image-version"}},
	}
	for _, s := range steps {
		var in io.Reader
		if s.stdin != "" {
			in = strings.NewReader(s.stdin)
		}
		if _, err := m.RT.Exec(ctx, TemplateName, in, s.argv...); err != nil {
			return fmt.Errorf("provision template (%s): %w", strings.Join(s.argv, " "), err)
		}
	}
	return nil
}

// Desktop is one agent desktop.
type Desktop struct {
	Name    string // user-facing name, such as "a"
	Machine string // runtime machine name, such as "adf-a"
	State   string // runtime state, such as "running" or "stopped"
}

// WarmTemplate makes sure the template runs as a branch source with its desktop up,
// so every branch starts with a ready desktop.
func (m *Manager) WarmTemplate(ctx context.Context) error {
	t, err := m.find(ctx, TemplateName)
	if err != nil {
		return err
	}
	if t == nil {
		return errors.New("no desktop template yet: run 'adf init' first")
	}
	if err := m.ensureBranchSource(ctx, t); err != nil {
		return err
	}
	if _, err := m.RT.Exec(ctx, TemplateName, nil, "desktop-start"); err != nil {
		return fmt.Errorf("start template desktop: %w", err)
	}
	return m.waitDesktopReady(ctx, TemplateName)
}

// waitDesktopReady polls until Firefox has a visible window. If the first poll already
// succeeds the desktop was warm and no settle time is added.
func (m *Manager) waitDesktopReady(ctx context.Context, machine string) error {
	deadline := time.Now().Add(m.ReadyTimeout)
	for attempt := 1; ; attempt++ {
		_, err := m.RT.Exec(ctx, machine, nil,
			"env", "DISPLAY="+image.Display, "xdotool", "search", "--onlyvisible", "--class", "firefox")
		if err == nil {
			if attempt == 1 {
				return nil
			}
			return sleepCtx(ctx, m.Settle)
		}
		if time.Now().After(deadline) {
			return fmt.Errorf("desktop in %s not ready after %s: %w", machine, m.ReadyTimeout, err)
		}
		if err := sleepCtx(ctx, m.Poll); err != nil {
			return err
		}
	}
}

func sleepCtx(ctx context.Context, d time.Duration) error {
	if d <= 0 {
		return nil
	}
	select {
	case <-ctx.Done():
		return ctx.Err()
	case <-time.After(d):
		return nil
	}
}

// CreateDesktop branches a new running desktop from the warm template.
func (m *Manager) CreateDesktop(ctx context.Context, name string) (Desktop, error) {
	if err := ValidateName(name); err != nil {
		return Desktop{}, err
	}
	machine := DesktopPrefix + name
	existing, err := m.find(ctx, machine)
	if err != nil {
		return Desktop{}, err
	}
	if existing != nil {
		return Desktop{}, fmt.Errorf("desktop %q already exists", name)
	}
	if err := m.WarmTemplate(ctx); err != nil {
		return Desktop{}, err
	}
	err = m.RT.Branch(ctx, smolvm.BranchOpts{
		From: TemplateName, Name: machine, Labels: map[string]string{LabelRole: RoleDesktop},
	})
	if err != nil {
		return Desktop{}, fmt.Errorf("branch desktop %q: %w", name, err)
	}
	return Desktop{Name: name, Machine: machine, State: "running"}, nil
}

// List returns all desktops sorted by name.
func (m *Manager) List(ctx context.Context) ([]Desktop, error) {
	ms, err := m.RT.List(ctx)
	if err != nil {
		return nil, err
	}
	var out []Desktop
	for _, mc := range ms {
		if mc.Labels[LabelRole] != RoleDesktop || !strings.HasPrefix(mc.Name, DesktopPrefix) {
			continue
		}
		out = append(out, Desktop{Name: strings.TrimPrefix(mc.Name, DesktopPrefix), Machine: mc.Name, State: mc.State})
	}
	sort.Slice(out, func(i, j int) bool { return out[i].Name < out[j].Name })
	return out, nil
}

func (m *Manager) mustFind(ctx context.Context, name string) (*smolvm.Machine, error) {
	if err := ValidateName(name); err != nil {
		return nil, err
	}
	mc, err := m.find(ctx, DesktopPrefix+name)
	if err != nil {
		return nil, err
	}
	if mc == nil {
		return nil, fmt.Errorf("no desktop named %q", name)
	}
	return mc, nil
}

// Stop stops a desktop.
func (m *Manager) Stop(ctx context.Context, name string) error {
	mc, err := m.mustFind(ctx, name)
	if err != nil {
		return err
	}
	return m.RT.Stop(ctx, mc.Name)
}

// Delete stops a desktop if it runs and removes it.
func (m *Manager) Delete(ctx context.Context, name string) error {
	mc, err := m.mustFind(ctx, name)
	if err != nil {
		return err
	}
	if mc.State == "running" {
		if err := m.RT.Stop(ctx, mc.Name); err != nil {
			return fmt.Errorf("stop %q: %w", name, err)
		}
	}
	return m.RT.Delete(ctx, mc.Name, false)
}

// Exec runs argv inside a desktop and returns its stdout.
func (m *Manager) Exec(ctx context.Context, name string, argv ...string) ([]byte, error) {
	return m.RT.Exec(ctx, DesktopPrefix+name, nil, argv...)
}

// Computer returns the computer-use actions for a desktop.
func (m *Manager) Computer(name string) *computer.Computer {
	machine := DesktopPrefix + name
	exec := func(ctx context.Context, stdin io.Reader, argv ...string) ([]byte, error) {
		return m.RT.Exec(ctx, machine, stdin, argv...)
	}
	return computer.New(exec, computer.Screen{Display: image.Display, Width: image.Width, Height: image.Height})
}

// app holds what commands need. It is built on first use so --help works without smolvm.
type app struct {

func (a *app) setup() error {

	}

	}

// findSmolvm returns the flag value, then smolvm on PATH, then the installer's default location.
func findSmolvm(flagValue string) (string, error) {
	if flagValue != "" {
		return flagValue, nil
	}
	if p, err := exec.LookPath("smolvm"); err == nil {
		return p, nil
	}
	if home, err := os.UserHomeDir(); err == nil {
		p := filepath.Join(home, ".smolvm", "smolvm")
		if _, err := os.Stat(p); err == nil {
			return p, nil
		}
	}
	return "", errors.New("smolvm not found: install it from https://github.com/smol-machines/smolvm or pass --smolvm PATH")
}

func newRootCmd() *cobra.Command {
	a := &app{}
	root := &cobra.Command{

import (

	"os"
	"os/exec"

// checkVirtualization reports whether this host can run hardware-accelerated VMs.
func checkVirtualization() error {
	switch runtime.GOOS {
	case "darwin":
		out, err := exec.Command("sysctl", "-n", "kern.hv_support").Output()
		if err != nil {
			return fmt.Errorf("sysctl kern.hv_support: %w", err)
		}
		if strings.TrimSpace(string(out)) != "1" {
			return errors.New("this Mac does not support Hypervisor.framework")
		}
		return nil
	case "linux":
		f, err := os.OpenFile("/dev/kvm", os.O_RDWR, 0)
		if err != nil {
			return fmt.Errorf("cannot open /dev/kvm (%v): enable KVM or nested virtualization and add your user to the kvm group", err)
		}
		return f.Close()
	default:
		return fmt.Errorf("%s hosts are not supported yet", runtime.GOOS)
	}
}

package main

import (
	"fmt"

)

// parseInts converts coordinate arguments to integers.
func parseInts(args ...string) ([]int, error) {
	out := make([]int, len(args))
	for i, s := range args {
		n, err := strconv.Atoi(s)
		if err != nil {
			return nil, fmt.Errorf("%q is not an integer", s)
		}
		out[i] = n
	}
	return out, nil
}

func TestFleetEndToEnd(t *testing.T) {
	ctx, cancel := context.WithTimeout(context.Background(), 15*time.Minute)
	defer cancel()
	m := fleet.NewManager(smolvm.New(smolvmBin(t), runner.Exec{}), os.Stderr)

		t.Fatal(err)
	}

	names := []string{"it-a", "it-b"}
	for _, n := range names {
		_ = m.Delete(ctx, n) // leftovers from an earlier failed run

			t.Fatal(err)
		}

		t.Cleanup(func() { _ = m.Delete(context.Background(), n) })

	ds, err := m.List(ctx)
	if err != nil {
		t.Fatal(err)
	}
	found := map[string]bool{}
	for _, d := range ds {
		found[d.Name] = d.State == "running"

	for _, n := range names {
		if !found[n] {
			t.Fatalf("desktop %s missing or not running in %+v", n, ds)
