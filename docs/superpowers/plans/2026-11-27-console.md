# Web Console Implementation Plan (plan 3 of 4)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** `adf console` opens a local web page showing every desktop live (a wall of screens with CPU, memory and the latest agent action), a focused view with mouse and keyboard takeover, and a feed of what agents and people did to each desktop.

**Architecture:** A new `internal/console` package serves a JSON API and embedded static files (vanilla HTML, CSS and JS, no build step). Live video is JPEG screenshots polled by the browser (measured: 44 ms and 71 KB per 1280x800 frame, about 120 frames/s across 6 desktops), deduplicated by a per-desktop frame cache. Takeover sends input events to the same `computer` package agents use. Metrics come from `smolvm machine stats` (about 0.5 s per call), refreshed by a background collector every 3 s. A new `internal/activity` package appends events as JSON lines to one shared file; every `adf mcp` process records its tool calls there and the console reads it, so agents in other processes appear live. API calls need a random bearer token that `adf console` puts in the URL it prints; the page moves it to `sessionStorage`.

**Tech Stack:** Go 1.25 (`net/http` method patterns, `embed`), go-sdk v1.8.0, vanilla JS, Barlow and Barlow Semi Condensed (SIL OFL, from `@fontsource` 5.3.0) embedded as woff2.

**Design direction (from the frontend-design pass):** a broadcast multiviewer. Each desktop is a monitor with a label strip (name, metrics) and a tally lamp whose colour means something: green ready, amber an agent acted in the last 3 s, red you have control or the last action failed, grey stopped. Palette: rack `#20262E`, panel `#2A313B`, line `#3A4350`, text `#E6E9EE`, muted `#9AA4B2`, amber `#F2A93B`, red `#E5484D`, green `#46A758`. One deliberate motion: the amber lamp glow, off under `prefers-reduced-motion`. No all-caps labels, no middle-dot meta strings, no monospace data labels.

**Facts this plan relies on (verified 2026-11-27):**
- `scrot -o -z -q 60 FILE.jpg` inside the guest writes a JPEG; median 44 ms per capture through `smolvm machine exec`, 71 KB at 1280x800; 6 parallel workers reach about 120 frames/s across 6 desktops.
- `smolvm machine stats --name N --json` returns `{"cpu_percent": 0.3, "memory": {"total_bytes", "used_bytes", "available_bytes", ...}, ...}` and takes about 0.53 s.
- Go 1.22+ `http.ServeMux` supports `"GET /api/desktops/{name}/screen"` patterns and `r.PathValue("name")`; `http.FileServerFS` serves an `fs.FS`.

rc=$?
rm -f "$f" "$f.` + ext + `"
exit $rc`

// ScreenshotJPEG returns the whole screen as JPEG bytes at quality 1 to 100.
// It is smaller and faster than Screenshot, which suits live views.
func (c *Computer) ScreenshotJPEG(ctx context.Context, quality int) ([]byte, error) {
	if quality < 1 || quality > 100 {
		return nil, fmt.Errorf("%w: jpeg quality %d out of range 1 to 100", ErrInvalid, quality)
	}
	script := captureScript("jpg", "-q "+itoa(quality)+" ")
	out, err := c.exec(ctx, nil, "env", "DISPLAY="+c.screen.Display, "sh", "-c", script)
	if err != nil {
		return nil, fmt.Errorf("screenshot: %w", err)
	}
	if !bytes.HasPrefix(out, jpegMagic) {
		return nil, fmt.Errorf("screenshot: output is not a JPEG (%d bytes)", len(out))
	}
	return out, nil
}

type MemoryStats struct {
	TotalBytes     int64 `json:"total_bytes"`
	UsedBytes      int64 `json:"used_bytes"`
	AvailableBytes int64 `json:"available_bytes"`
}

// Stats is a running machine's resource use, as seen inside the guest.
type Stats struct {
	CPUPercent float64     `json:"cpu_percent"`
	Memory     MemoryStats `json:"memory"`
}

// Stats samples a running machine's CPU and memory. It takes about half a second.
func (c *Client) Stats(ctx context.Context, name string) (Stats, error) {
	out, err := c.machine(ctx, nil, "stats", "--name", name, "--json")
	if err != nil {
		return Stats{}, err
	}
	var s Stats
	if err := json.Unmarshal(out, &s); err != nil {
		return Stats{}, fmt.Errorf("parse smolvm machine stats: %w", err)
	}
	return s, nil
}

```go
// Package activity records what agents and people do to desktops. Every adf process
// appends JSON lines to one shared file, and the console reads it to show a live feed.
package activity

import (
	"bytes"
	"encoding/json"
	"errors"
	"io"
	"io/fs"
	"os"
	"path/filepath"
	"sync"
	"time"
)

// Sources of events.
const (
	SourceAgent   = "agent"   // an MCP tool call
	SourceConsole = "console" // a person using the web console
)

// Event is one action on one desktop.
type Event struct {
	Time       time.Time `json:"time"`
	Desktop    string    `json:"desktop"`
	Source     string    `json:"source"`
	Action     string    `json:"action"`
	Detail     string    `json:"detail,omitempty"`
	OK         bool      `json:"ok"`
	Error      string    `json:"error,omitempty"`
	DurationMS int64     `json:"duration_ms"`
}

// Recorder receives events. Recording must never make an action fail.
type Recorder interface {
	Record(Event)
}

// DefaultMaxSize is the size at which Log moves the file to Path+".1" and starts a new one.
const DefaultMaxSize = 8 << 20

// maxRead caps how much of the file a single read looks at.
const maxRead = 4 << 20

// DefaultPath returns the shared log location: ~/Library/Application Support/adf/activity.jsonl
// on macOS and ~/.config/adf/activity.jsonl on Linux.
func DefaultPath() (string, error) {
	dir, err := os.UserConfigDir()
	if err != nil {
		return "", err
	}
	return filepath.Join(dir, "adf", "activity.jsonl"), nil
}

// Log appends events to a JSON-lines file. It is safe for concurrent use, and separate
// processes may append to the same file.
type Log struct {
	Path    string
	MaxSize int64 // 0 means DefaultMaxSize

	mu sync.Mutex
}

// Record implements Recorder. Errors are dropped on purpose.
func (l *Log) Record(e Event) { _ = l.Append(e) }

// Append writes one event as a line. The file is private to the user because it can
// contain commands that agents ran.
func (l *Log) Append(e Event) error {
	line, err := json.Marshal(e)
	if err != nil {
		return err
	}
	l.mu.Lock()
	defer l.mu.Unlock()
	if err := os.MkdirAll(filepath.Dir(l.Path), 0o700); err != nil {
		return err
	}
	max := l.MaxSize
	if max <= 0 {
		max = DefaultMaxSize
	}
	if fi, err := os.Stat(l.Path); err == nil && fi.Size() >= max {
		_ = os.Rename(l.Path, l.Path+".1")
	}
	f, err := os.OpenFile(l.Path, os.O_APPEND|os.O_CREATE|os.O_WRONLY, 0o600)
	if err != nil {
		return err
	}
	_, werr := f.Write(append(line, '\n'))
	cerr := f.Close()
	if werr != nil {
		return werr
	}
	return cerr
}

// ReadFrom returns the complete events after byte offset cursor and the cursor for the next
// call. A trailing partial line is left for later. If the file is now shorter than cursor
// (it was rotated), reading restarts at the beginning.
func ReadFrom(path string, cursor int64) ([]Event, int64, error) {
	f, err := os.Open(path)
	if errors.Is(err, fs.ErrNotExist) {
		return nil, 0, nil
	}
	if err != nil {
		return nil, cursor, err
	}
	defer f.Close()
	fi, err := f.Stat()
	if err != nil {
		return nil, cursor, err
	}
	if fi.Size() < cursor {
		cursor = 0
	}
	if _, err := f.Seek(cursor, io.SeekStart); err != nil {
		return nil, cursor, err
	}
	data, err := io.ReadAll(io.LimitReader(f, maxRead))
	if err != nil {
		return nil, cursor, err
	}
	end := bytes.LastIndexByte(data, '\n')
	if end < 0 {
		return nil, cursor, nil
	}
	return parse(data[:end]), cursor + int64(end) + 1, nil
}

// Recent returns up to n of the newest events and the cursor just after them.
func Recent(path string, n int) ([]Event, int64, error) {
	f, err := os.Open(path)
	if errors.Is(err, fs.ErrNotExist) {
		return nil, 0, nil
	}
	if err != nil {
		return nil, 0, err
	}
	defer f.Close()
	fi, err := f.Stat()
	if err != nil {
		return nil, 0, err
	}
	start := fi.Size() - maxRead
	if start < 0 {
		start = 0
	}
	if _, err := f.Seek(start, io.SeekStart); err != nil {
		return nil, 0, err
	}
	data, err := io.ReadAll(io.LimitReader(f, maxRead))
	if err != nil {
		return nil, 0, err
	}
	end := bytes.LastIndexByte(data, '\n')
	if end < 0 {
		return nil, start, nil
	}
	body := data[:end]
	if start > 0 {
		// The first line may have been cut in half; skip it.
		if i := bytes.IndexByte(body, '\n'); i >= 0 {
			body = body[i+1:]
		} else {
			body = nil
		}
	}
	events := parse(body)
	if len(events) > n {
		events = events[len(events)-n:]
	}
	return events, start + int64(end) + 1, nil
}

func parse(data []byte) []Event {
	var out []Event
	for _, line := range bytes.Split(data, []byte{'\n'}) {
		if len(line) == 0 {
			continue
		}
		var e Event
		if json.Unmarshal(line, &e) == nil {
			out = append(out, e)
		}
	}
	return out
}

// Package httpauth protects HTTP handlers with a static bearer token.
package httpauth

import (
	"crypto/rand"
	"crypto/subtle"
	"encoding/hex"
	"net/http"
	"strings"
)

// RequireToken rejects requests without "Authorization: Bearer <token>".
func RequireToken(token string, next http.Handler) http.Handler {
	want := []byte(token)
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		scheme, got, ok := strings.Cut(r.Header.Get("Authorization"), " ")
		if !ok || !strings.EqualFold(scheme, "Bearer") || subtle.ConstantTimeCompare([]byte(got), want) != 1 {
			w.Header().Set("WWW-Authenticate", "Bearer")
			http.Error(w, "unauthorized", http.StatusUnauthorized)
			return
		}
		next.ServeHTTP(w, r)
	})
}

// NewToken returns 32 random bytes as hex.
func NewToken() (string, error) {
	b := make([]byte, 32)
	if _, err := rand.Read(b); err != nil {
		return "", err
	}
	return hex.EncodeToString(b), nil
}

// recorded wraps a tool handler so each call is written to the activity log.
func recorded[In, Out any](t *tools, action string, desktop func(In) string, detail func(In) string, h mcp.ToolHandlerFor[In, Out]) mcp.ToolHandlerFor[In, Out] {
	return func(ctx context.Context, req *mcp.CallToolRequest, in In) (*mcp.CallToolResult, Out, error) {
		start := time.Now()
		res, out, err := h(ctx, req, in)
		if t.opts.Activity != nil {
			e := activity.Event{Time: start, Desktop: desktop(in), Source: activity.SourceAgent, Action: action,

			if detail != nil {
				e.Detail = detail(in)
			}
			if err != nil {
				e.Error = err.Error()
			}
			t.opts.Activity.Record(e)
		}
		return res, out, err
	}
}

import (

// inputReq is one input event from the browser.
type inputReq struct {
	Kind   string `json:"kind"` // click, double_click, move, drag, scroll, type, key
	X      int    `json:"x"`
	Y      int    `json:"y"`
	ToX    int    `json:"to_x"`
	ToY    int    `json:"to_y"`
	Button string `json:"button"`
	Dir    string `json:"dir"`
	Amount int    `json:"amount"`
	Text   string `json:"text"`
	Key    string `json:"key"` // KeyboardEvent.key
	Ctrl   bool   `json:"ctrl"`
	Alt    bool   `json:"alt"`
	Shift  bool   `json:"shift"`
	Meta   bool   `json:"meta"`
}

func point(x, y int) string { return fmt.Sprintf("(%d, %d)", x, y) }

// applyInput performs one input event and returns a short description for the activity log.
// Typed text is described by its length only.
func applyInput(ctx context.Context, c *computer.Computer, in inputReq) (string, error) {
	switch in.Kind {
	case "click":
		return point(in.X, in.Y), c.Click(ctx, in.X, in.Y, computer.Button(in.Button))
	case "double_click":
		return point(in.X, in.Y), c.DoubleClick(ctx, in.X, in.Y, computer.Button(in.Button))
	case "move":
		return point(in.X, in.Y), c.Move(ctx, in.X, in.Y)
	case "drag":
		return point(in.X, in.Y) + " to " + point(in.ToX, in.ToY), c.Drag(ctx, in.X, in.Y, in.ToX, in.ToY)
	case "scroll":
		amount := in.Amount
		if amount == 0 {
			amount = 1
		}
		return fmt.Sprintf("%s %d at %s", in.Dir, amount, point(in.X, in.Y)),
			c.Scroll(ctx, in.X, in.Y, computer.Direction(in.Dir), amount)
	case "type":
		return fmt.Sprintf("%d characters", len([]rune(in.Text))), c.Type(ctx, in.Text)
	case "key":
		combo, text, err := browserKey(in.Key, in.Ctrl, in.Alt, in.Shift, in.Meta)
		if err != nil {
			return in.Key, err
		}
		if text != "" {
			return "1 character", c.Type(ctx, text)
		}
		return combo, c.Key(ctx, combo)
	}
	return "", fmt.Errorf("%w: unknown input kind %q", computer.ErrInvalid, in.Kind)
}

// specialKeys maps KeyboardEvent.key names to xdotool key names.
var specialKeys = map[string]string{
	"Enter": "Return", "Backspace": "BackSpace", "Tab": "Tab", "Escape": "Escape", "Delete": "Delete",
	"Insert": "Insert", "Home": "Home", "End": "End", "PageUp": "Page_Up", "PageDown": "Page_Down",
	"ArrowUp": "Up", "ArrowDown": "Down", "ArrowLeft": "Left", "ArrowRight": "Right", " ": "space",
}

func init() {
	for i := 1; i <= 12; i++ {
		k := "F" + strconv.Itoa(i)
		specialKeys[k] = k
	}
}

// browserKey turns a browser key event into an xdotool chord, or into text to type when it
// is a plain character (the browser has already applied Shift to it).
func browserKey(key string, ctrl, alt, shift, meta bool) (combo, text string, err error) {
	name, special := specialKeys[key]
	chord := ctrl || alt || meta
	if !special {
		r := []rune(key)
		if len(r) != 1 {
			return "", "", fmt.Errorf("%w: unsupported key %q", computer.ErrInvalid, key)
		}
		if !chord {
			return "", key, nil
		}
		if r[0] > unicode.MaxASCII || !(unicode.IsLetter(r[0]) || unicode.IsDigit(r[0])) {
			return "", "", fmt.Errorf("%w: unsupported shortcut key %q", computer.ErrInvalid, key)
		}
		name = strings.ToLower(key)
	}
	var parts []string
	if ctrl {
		parts = append(parts, "ctrl")
	}
	if alt {
		parts = append(parts, "alt")
	}
	if shift && (special || chord) {
		parts = append(parts, "shift")
	}
	if meta {
		parts = append(parts, "super")
	}
	return strings.Join(append(parts, name), "+"), "", nil
}

//go:embed web
var webFS embed.FS

// Options configures the console.
type Options struct {
	Token        string            // required on every /api/ request
	ActivityPath string            // activity log to show; empty disables the feed
	Recorder     activity.Recorder // records console actions; nil disables recording
	FrameMaxAge  time.Duration     // reuse a captured frame for this long (default 100 ms)
	JPEGQuality  int               // default 60
	StatsEvery   time.Duration     // metrics refresh period (default 3 s)
}

// Server is the console backend.
type Server struct {
	fleet  *fleet.Manager
	opts   Options
	frames frameCache

	statsMu sync.Mutex
	stats   map[string]desktopStats
}

type desktopStats struct {
	CPUPercent    float64 `json:"cpu_percent"`
	MemUsedBytes  int64   `json:"mem_used_bytes"`
	MemTotalBytes int64   `json:"mem_total_bytes"`
}

type desktopJSON struct {
	Name  string        `json:"name"`
	State string        `json:"state"`
	Stats *desktopStats `json:"stats,omitempty"`
}

// New returns a console for f.
func New(f *fleet.Manager, opts Options) *Server {
	if opts.FrameMaxAge == 0 {
		opts.FrameMaxAge = 100 * time.Millisecond
	}
	if opts.JPEGQuality == 0 {
		opts.JPEGQuality = 60
	}
	if opts.StatsEvery == 0 {
		opts.StatsEvery = 3 * time.Second
	}
	return &Server{fleet: f, opts: opts, frames: frameCache{entries: map[string]*frameEntry{}}, stats: map[string]desktopStats{}}
}

// Handler serves the static page without a token and the API with one.
func (s *Server) Handler() http.Handler {
	api := http.NewServeMux()
	api.HandleFunc("GET /api/status", s.handleStatus)
	api.HandleFunc("GET /api/desktops", s.handleList)
	api.HandleFunc("POST /api/desktops", s.handleCreate)
	api.HandleFunc("DELETE /api/desktops/{name}", s.handleDelete)
	api.HandleFunc("GET /api/desktops/{name}/screen", s.handleScreen)
	api.HandleFunc("POST /api/desktops/{name}/input", s.handleInput)
	api.HandleFunc("GET /api/activity", s.handleActivity)

	web, err := fs.Sub(webFS, "web")
	if err != nil {
		panic(err) // the embedded directory always exists
	}
	root := http.NewServeMux()
	root.Handle("/api/", httpauth.RequireToken(s.opts.Token, api))
	root.Handle("/", http.FileServerFS(web))
	return root
}

// RunCollector refreshes desktop metrics until ctx is done.
func (s *Server) RunCollector(ctx context.Context) {
	ticker := time.NewTicker(s.opts.StatsEvery)
	defer ticker.Stop()
	for {
		s.refreshStats(ctx)
		select {
		case <-ctx.Done():
			return
		case <-ticker.C:
		}
	}
}

// refreshStats samples every running desktop in parallel (each sample takes about 0.5 s).
func (s *Server) refreshStats(ctx context.Context) {
	ds, err := s.fleet.List(ctx)
	if err != nil {
		return
	}
	fresh := map[string]desktopStats{}
	var mu sync.Mutex
	var wg sync.WaitGroup
	sem := make(chan struct{}, 8)
	for _, d := range ds {
		if d.State != "running" {
			continue
		}
		wg.Add(1)
		go func(name string) {
			defer wg.Done()
			sem <- struct{}{}
			defer func() { <-sem }()
			st, err := s.fleet.Stats(ctx, name)
			if err != nil {
				return
			}
			mu.Lock()
			fresh[name] = desktopStats{CPUPercent: st.CPUPercent, MemUsedBytes: st.Memory.UsedBytes, MemTotalBytes: st.Memory.TotalBytes}
			mu.Unlock()
		}(d.Name)
	}
	wg.Wait()
	s.statsMu.Lock()
	s.stats = fresh
	s.statsMu.Unlock()
}

func writeJSON(w http.ResponseWriter, status int, v any) {
	w.Header().Set("Content-Type", "application/json")
	w.WriteHeader(status)
	_ = json.NewEncoder(w).Encode(v)
}

func writeMessage(w http.ResponseWriter, status int, msg string) {
	writeJSON(w, status, map[string]string{"error": msg})
}

// writeErr maps errors to status codes the page can act on.
func writeErr(w http.ResponseWriter, err error) {
	status := http.StatusInternalServerError
	switch {
	case errors.Is(err, computer.ErrInvalid), errors.Is(err, fleet.ErrInvalidName):
		status = http.StatusBadRequest
	case errors.Is(err, fleet.ErrNotFound), smolvm.IsNotFound(err):
		status = http.StatusNotFound
	case errors.Is(err, fleet.ErrExists):
		status = http.StatusConflict
	}
	writeMessage(w, status, err.Error())
}

func (s *Server) record(desktop, action, detail string, start time.Time, err error) {
	if s.opts.Recorder == nil {
		return
	}
	e := activity.Event{Time: start, Desktop: desktop, Source: activity.SourceConsole, Action: action, Detail: detail,
		OK: err == nil, DurationMS: time.Since(start).Milliseconds()}
	if err != nil {
		e.Error = err.Error()
	}
	s.opts.Recorder.Record(e)
}

	state, err := s.fleet.TemplateState(r.Context())
	if err != nil {
		writeErr(w, err)
		return
	}
	writeJSON(w, http.StatusOK, map[string]string{"template": state})

func (s *Server) handleList(w http.ResponseWriter, r *http.Request) {

	if err != nil {
		writeErr(w, err)
		return

	out := []desktopJSON{}
	s.statsMu.Lock()
	for _, d := range ds {
		j := desktopJSON{Name: d.Name, State: d.State}

	s.statsMu.Unlock()
	writeJSON(w, http.StatusOK, map[string]any{"desktops": out})
}

	if err := fleet.ValidateName(name); err != nil {
		writeErr(w, err)
		return
	}
	var in inputReq
	if err := json.NewDecoder(http.MaxBytesReader(w, r.Body, 1<<16)).Decode(&in); err != nil {
		writeMessage(w, http.StatusBadRequest, "send one JSON input event")
		return
	}
	start := time.Now()
	detail, err := applyInput(r.Context(), s.fleet.Computer(name), in)

func (s *Server) handleActivity(w http.ResponseWriter, r *http.Request) {
	if s.opts.ActivityPath == "" {
		writeJSON(w, http.StatusOK, map[string]any{"events": []activity.Event{}, "cursor": 0})
		return
	}
	var (
		events []activity.Event
		cursor int64
		err    error
	)
	if c := r.URL.Query().Get("cursor"); c != "" {
		n, perr := strconv.ParseInt(c, 10, 64)
		if perr != nil || n < 0 {
			writeMessage(w, http.StatusBadRequest, "cursor must be a non-negative integer")
			return
		}
		events, cursor, err = activity.ReadFrom(s.opts.ActivityPath, n)
	} else {
		events, cursor, err = activity.Recent(s.opts.ActivityPath, 200)
	}
	if err != nil {
		writeErr(w, err)
		return
	}
	if events == nil {
		events = []activity.Event{}
	}
	writeJSON(w, http.StatusOK, map[string]any{"events": events, "cursor": cursor})
}

// frameCache keeps the latest frame per desktop so several viewers share one capture.
type frameCache struct {
	mu      sync.Mutex
	entries map[string]*frameEntry
}

type frameEntry struct {
	mu   sync.Mutex // held during a capture, so concurrent requests wait for it
	data []byte
	at   time.Time
}

func (c *frameCache) get(ctx context.Context, name string, maxAge time.Duration, capture func(context.Context) ([]byte, error)) ([]byte, error) {
	c.mu.Lock()
	e := c.entries[name]
	if e == nil {
		e = &frameEntry{}
		c.entries[name] = e
	}
	c.mu.Unlock()

	e.mu.Lock()
	defer e.mu.Unlock()
	if e.data != nil && time.Since(e.at) < maxAge {
		return e.data, nil
	}
	data, err := capture(ctx)
	if err != nil {
		return nil, err
	}
	e.data, e.at = data, time.Now()
	return data, nil
}

func (c *frameCache) forget(name string) {
	c.mu.Lock()
	delete(c.entries, name)
	c.mu.Unlock()
}

// Frames: each view fetches the next frame only after the previous one arrives.

class FrameLoop {
  constructor(img, name, interval) {
    this.img = img;
    this.name = name;
    this.interval = interval;
    this.running = false;
    this.url = null;
  }
  start() {
    if (this.running) return;
    this.running = true;
    this.tick();
  }
  stop() {
    this.running = false;
    clearTimeout(this.timer);
  }
  async tick() {
    if (!this.running) return;
    const began = performance.now();
    const monitor = this.img.parentElement;
    if (isRunning(this.name)) {
      try {
        const res = await api(`/api/desktops/${encodeURIComponent(this.name)}/screen`);
        const url = URL.createObjectURL(await res.blob());
        const prev = this.url;
        this.url = url;
        this.img.onload = () => { if (prev) URL.revokeObjectURL(prev); };
        this.img.src = url;
        monitor.classList.remove("stale");
      } catch (_) {
        monitor.classList.add("stale");
      }
    } else {
      monitor.classList.add("stale");
    }
    if (!this.running) return;
    const wait = Math.max(0, this.interval - (performance.now() - began));
    this.timer = setTimeout(() => this.tick(), wait);
  }
}

function toScreen(ev) {
  const r = $("focus-screen").getBoundingClientRect();
  const scale = Math.min(r.width / SCREEN_W, r.height / SCREEN_H);

  const y = Math.round((ev.clientY - top) / scale);

  return { x, y };
}

package main

import (
	"context"
	"errors"
	"fmt"
	"net"
	"net/http"
	"os/exec"
	"runtime"
	"time"

	"github.com/spf13/cobra"

	"github.com/CodeGeek04/agent-desktop-fleet/internal/console"
	"github.com/CodeGeek04/agent-desktop-fleet/internal/httpauth"
)

func consoleCommand(a *app) *cobra.Command {
	var addr string
	var noOpen bool
	cmd := &cobra.Command{
		Use:   "console",
		Short: "Open the web console: every desktop live, takeover and agent activity",
		Args:  cobra.NoArgs,
		RunE: func(cmd *cobra.Command, _ []string) error {
			if err := a.setup(); err != nil {
				return err
			}
			token, err := httpauth.NewToken()
			if err != nil {
				return err
			}
			opts := console.Options{Token: token}
			if log := activityLog(); log != nil {
				opts.ActivityPath = log.Path
				opts.Recorder = log
			}
			srv := console.New(a.fleet, opts)

			ln, err := net.Listen("tcp", addr)
			if err != nil {
				return fmt.Errorf("listen on %s: %w (pass --addr to use another port)", addr, err)
			}
			url := fmt.Sprintf("http://%s/?t=%s", ln.Addr(), token)
			fmt.Fprintf(cmd.OutOrStdout(), "adf console: %s\nPress Ctrl+C to stop.\n", url)
			if host, _, _ := net.SplitHostPort(ln.Addr().String()); !net.ParseIP(host).IsLoopback() {
				fmt.Fprintln(cmd.ErrOrStderr(), "warning: the console is reachable from other machines; anyone with the link controls your desktops")
			}
			if !noOpen {
				openBrowser(url)
			}

			ctx := cmd.Context()
			go srv.RunCollector(ctx)
			httpSrv := &http.Server{Handler: srv.Handler(), ReadHeaderTimeout: 10 * time.Second}
			go func() {
				<-ctx.Done()
				shutdownCtx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
				defer cancel()
				_ = httpSrv.Shutdown(shutdownCtx)
			}()
			if err := httpSrv.Serve(ln); err != nil && !errors.Is(err, http.ErrServerClosed) {
				return err
			}
			return nil
		},
	}
	cmd.Flags().StringVar(&addr, "addr", "127.0.0.1:7300", "address to serve the console on")
	cmd.Flags().BoolVar(&noOpen, "no-open", false, "do not open a browser")
	return cmd
}

// openBrowser opens url in the default browser; failures are ignored because the URL is printed.
func openBrowser(url string) {
	var cmd *exec.Cmd
	switch runtime.GOOS {
	case "darwin":
		cmd = exec.Command("open", url)
	case "linux":
		cmd = exec.Command("xdg-open", url)
	default:
		return
	}
	_ = cmd.Start()
}
```

func TestConsoleEndToEnd(t *testing.T) {
	ctx, cancel := context.WithTimeout(context.Background(), 15*time.Minute)
	defer cancel()
	m := fleet.NewManager(smolvm.New(smolvmBin(t), runner.Exec{}), os.Stderr)
	logPath := filepath.Join(t.TempDir(), "activity.jsonl")
	srv := console.New(m, console.Options{Token: "it", ActivityPath: logPath, Recorder: &activity.Log{Path: logPath}})
	go srv.RunCollector(ctx)
	ts := httptest.NewServer(srv.Handler())
	defer ts.Close()

	req := func(method, path, body string) (*http.Response, []byte) {
		t.Helper()
		r, err := http.NewRequestWithContext(ctx, method, ts.URL+path, strings.NewReader(body))
		if err != nil {
			t.Fatal(err)
		}
		r.Header.Set("Authorization", "Bearer it")
		res, err := http.DefaultClient.Do(r)
		if err != nil {
			t.Fatal(err)
		}
		defer res.Body.Close()
		b, _ := io.ReadAll(res.Body)
		return res, b
	}

	_ = m.Delete(ctx, "con-it")
	if res, b := req("POST", "/api/desktops", `{"name":"con-it"}`); res.StatusCode != http.StatusCreated {
		t.Fatalf("create: %d %s", res.StatusCode, b)
	}
	t.Cleanup(func() { _ = m.Delete(context.Background(), "con-it") })

	res, before := req("GET", "/api/desktops/con-it/screen", "")
	if res.StatusCode != http.StatusOK || !bytes.HasPrefix(before, []byte{0xFF, 0xD8, 0xFF}) {
		t.Fatalf("screen: %d, %d bytes", res.StatusCode, len(before))
	}
	for _, in := range []string{`{"kind":"key","key":"l","ctrl":true}`, `{"kind":"type","text":"about:robots"}`, `{"kind":"key","key":"Enter"}`} {
		if res, b := req("POST", "/api/desktops/con-it/input", in); res.StatusCode != http.StatusNoContent {
			t.Fatalf("input %s: %d %s", in, res.StatusCode, b)
		}
	}
	time.Sleep(2 * time.Second)
	if _, after := req("GET", "/api/desktops/con-it/screen", ""); bytes.Equal(before, after) {
		t.Fatal("screen did not change after typing a URL")
	}

	deadline := time.Now().Add(15 * time.Second)
	for {
		_, b := req("GET", "/api/desktops", "")
		var out struct {
			Desktops []struct {
				Name  string
				Stats *struct {
					MemTotalBytes int64 `json:"mem_total_bytes"`
				}
			}
		}
		if err := json.Unmarshal(b, &out); err != nil {
			t.Fatal(err)
		}
		ok := false
		for _, d := range out.Desktops {
			if d.Name == "con-it" && d.Stats != nil && d.Stats.MemTotalBytes > 0 {
				ok = true
			}
		}
		if ok {
			break
		}
		if time.Now().After(deadline) {
			t.Fatalf("no stats for con-it after 15 s: %s", b)
		}
		time.Sleep(500 * time.Millisecond)
	}

	_, b := req("GET", "/api/activity", "")
	var feed struct{ Events []activity.Event }
	if err := json.Unmarshal(b, &feed); err != nil {
		t.Fatal(err)
	}
	if len(feed.Events) < 4 {
		t.Fatalf("want create plus three inputs in the feed, got %+v", feed.Events)
	}
}
